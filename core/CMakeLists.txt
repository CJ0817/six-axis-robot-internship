cmake_minimum_required(VERSION 3.20)
project(robot_core_only VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(robot_contract SHARED "${ROOT}/src/common/abi_probe.c" "${ROOT}/src/common/abi_v1.c" "${ROOT}/src/kinematics/forward.c" "${ROOT}/src/kinematics/inverse.c")
add_library(robot_planning SHARED "${ROOT}/src/planning/joint.c" "${ROOT}/src/planning/cartesian.c")
add_library(robot_control SHARED "${ROOT}/src/control/pid.c")
foreach(target robot_contract robot_planning robot_control)
  target_include_directories(${target} PUBLIC "${ROOT}/include")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(${target} PRIVATE m)
  install(TARGETS ${target} LIBRARY DESTINATION lib)
endforeach()
install(DIRECTORY "${ROOT}/include/" DESTINATION include FILES_MATCHING PATTERN "*.h")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_api_demo "${ROOT}/examples/core_api_demo.c")
  target_link_libraries(core_api_demo PRIVATE robot_contract robot_planning robot_control m)
  target_compile_options(core_api_demo PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME pure_c_interface_example COMMAND core_api_demo)
endif()
